cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2)
include(pico_sdk_import.cmake)
project(pulse_generator C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(pulse_generator main.c)
pico_generate_pio_header(pulse_generator ${CMAKE_CURRENT_LIST_DIR}/pulse_generator.pio)
target_link_libraries(pulse_generator pico_stdlib hardware_pio hardware_dma hardware_clocks)
pico_enable_stdio_usb(pulse_generator 1)
pico_enable_stdio_uart(pulse_generator 0)
pico_add_extra_outputs(pulse_generator)

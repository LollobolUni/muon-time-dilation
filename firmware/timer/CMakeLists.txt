cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2)
include(pico_sdk_import.cmake)
project(timer C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(timer main.c)
pico_generate_pio_header(timer ${CMAKE_CURRENT_LIST_DIR}/timer.pio)
target_link_libraries(timer pico_stdlib hardware_pio hardware_dma hardware_clocks)
pico_enable_stdio_usb(timer 1)
pico_enable_stdio_uart(timer 0)
pico_add_extra_outputs(timer)
